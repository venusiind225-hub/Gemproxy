# Copy this file to .env for local development.
# NEVER commit a real .env file — only .env.example belongs in git.
# On Railway, set these under your service's "Variables" tab instead of in a file.

# --- API keys: both now OPTIONAL. See middleware/auth.js for full detail. ---
#
# By default this proxy is bring-your-own-key: whatever you paste into
# Janitor's normal "API key" field IS your real Gemini key, forwarded
# through untouched. You can leave BOTH of the vars below unset — nothing
# needs to be configured here at all, and rotating keys means only ever
# touching Janitor, never Railway.
#
# Multiple comma-separated keys are supported both here and in Janitor's
# API key field (e.g. key1, key2, key3). Supports both newer Google
# authorization keys (AQ...) and legacy keys (AIza...).
# When an active key runs out (HTTP 429 / quota exhausted), the proxy
# automatically cycles to the next key and retries seamlessly.
#
# GEMINI_API_KEY is only needed if you want either of:
#   - a server-side fallback so curl/testing works with no client key, or
#   - "password mode" below.
# GEMINI_API_KEY=AQ.your-first-key,AQ.your-second-key

# PROXY_API_KEY: optional "password mode". If BOTH this and GEMINI_API_KEY
# are set, typing PROXY_API_KEY into Janitor's field (instead of your real
# key) makes the proxy substitute in the real GEMINI_API_KEY server-side —
# useful if you don't want your real key visible in Janitor's own settings.
# Generate one with: openssl rand -hex 32
# PROXY_API_KEY=your-own-long-random-secret

# --- Optional overrides (sensible defaults are used if omitted) ---

# Upstream API base URL (rarely needs changing).
# GEMINI_BASE_URL=https://generativelanguage.googleapis.com

# Comma-separated list of browser origins allowed to call this proxy directly.
# Leave unset if you only call it from server-side code, scripts, or curl.
# ALLOWED_ORIGINS=https://your-frontend.example.com

# Rate limiting (per IP) — a safety net against a runaway bug/loop burning
# through your quota, NOT a quota raiser. Defaults are generous already
# (180 requests per 60s) to comfortably cover normal interactive roleplay
# (regenerates, swipes, streaming reconnects). Raise further if needed.
# RATE_LIMIT_WINDOW_MS=60000
# RATE_LIMIT_MAX=180

# Retries on transient Gemini errors (429 rate-limited / 500 / 503
# overloaded / 504) before giving up. Only ever retries the INITIAL
# request — once a stream has started relaying to the client it's never
# retried mid-flight.
# RETRY_ATTEMPTS=3
# RETRY_BASE_DELAY_MS=500

# Cooldown duration (in ms) when an API key hits rate limits or quota (429)
# before the proxy considers attempting it again. Defaults to 60000 (1 minute).
# KEY_COOLDOWN_MS=60000

# Default max_tokens applied ONLY when the client (Janitor) doesn't send
# its own max_tokens. Whatever Janitor sends always wins over this.
# Defaults to 12000 even if you don't set this var at all.
# DEFAULT_MAX_TOKENS=12000

# Same "only if the client didn't already send one" rule as above, but
# these default to unset (i.e. Janitor's temperature/top_p always used).
# DEFAULT_TEMPERATURE=1
# DEFAULT_TOP_P=0.95

# Hard ceiling on max_tokens — applied even if the client asks for more.
# Unset by default, i.e. no ceiling beyond what the client/model allow.
# MAX_TOKENS_CAP=32000

# Gemini "thinking" (extended reasoning). Applied only when the client
# didn't already ask for thinking itself on that request.

# Master switch — turn on to have this proxy set a reasoning level itself.
# If left off, Gemini just uses its own default thinking behavior per model.
# ENABLE_THINKING=true

# Only used when ENABLE_THINKING=true. minimal | low | medium | high.
# For Gemini 3.x models (gemini-3-flash, gemini-3.6-flash, gemini-3.7-flash,
# gemini-3.8-flash, ...) this is sent directly as the model's thinking level.
# Note: gemini-3.7-flash and gemini-3.8-flash reject "minimal" with a 400 —
# use "low" as the lightest setting for those two.
# For older Gemini 2.5 models it's mapped to an approximate token budget
# instead, since those models don't understand levels.
# REASONING_EFFORT=low

# Whether to ask Gemini for its thought summaries. Off by default.
# Independent of ENABLE_THINKING — most current models still reason a bit
# even without an explicit level set, so this can be turned on by itself.
# When on, thoughts are wrapped in <think></think> tags and prepended
# to the reply (or, when streaming, opened/closed exactly once around
# however many SSE chunks the reasoning spans).
# SHOW_REASONING=false

# --- Optional roleplay/jailbreak tricks — all OFF by default ---
# Each of these changes how Gemini responds, so nothing activates unless
# you explicitly set it to true. Test one at a time.

# Appends a trailing assistant turn telling Gemini explicit/NSFW content is
# allowed and refusal phrasing is off-limits.
# ENABLE_PREFILL=false

# Appends a fake "(OOC: Continue?)" / "(OOC: Yes)" turn pair before
# generation — a continuation-priming trick to nudge past refusals.
# ENABLE_OOC_TRICK=false

# Replaces every space in the conversation with an invisible Braille blank
# character (U+2800), defeating keyword-based content filters. Changes the
# literal text Gemini sees on every turn.
# ENABLE_BRAILLE_TRICK=false

# Collapses the whole conversation into a single assistant-role turn
# formatted as a script (Name: line) instead of alternating turns.
# ENABLE_NOASS=false

# Gives Gemini a Google Search tool it can call mid-generation for
# real-world facts/grounding.
# ENABLE_GOOGLE_SEARCH=false

# Port to listen on locally (Railway injects its own PORT automatically).
# PORT=8080
